// Initial model and data backend structure designed with assistance from ChatGPT.

Firestore:
users/{userId}

friendships/{friendshipId} // later feature

tripInvitations/{token}      // written and read only by Cloud Functions, never by clients
    uid: string                // same as the document ID and the token (22 random URL-safe chars)
    tripId: string
    inviterUserId: string      // the trip owner
    inviteeUserId: string | null  // null for a link anyone can use
    status: "PENDING" | "REVOKED"  // PENDING until revoked or expired
    creationTime: timestamp
    expirationTime: timestamp  // 7 days after creation

trips/{tripId}
    name: string
    description: string | null
    destinations: array<{name: string, latitude: number, longitude: number, countryCode?: string}>  // ordered
                               // 1 to 10 entries, name not blank,
                               // latitude in [-90, 90], longitude in [-180, 180]
                               // countryCode optional, ISO 3166-1 alpha-2 format (e.g. "PT", "FR")
                               // existing trips without countryCode remain valid
    startDate: timestamp
    endDate: timestamp         // >= startDate, stored even when flexibleDates is true
    flexibleDates: bool
    travelMode: "SOLO" | "GROUP"
    travelerCount: int         // SOLO: exactly 1, GROUP: 2 to 10
    ownerId: string            // auth uid of the creator, authoritative owner
    memberIds: array<string>   // contains ownerId at creation
    status: "PLANNED" | "ACTIVE" | "COMPLETED"
    createdAt: timestamp
    members/{userId}
        userId: string
        role: "OWNER" | "EDITOR" | "VIEWER"
        joinTime: timestamp
    activities/{activityId}
        activityType: "SUGGESTION" | "ACCEPTED"  // clients only create "ACCEPTED" for now
        name: string               // not blank, at most 100 characters
        description: string | null  // at most 2000 characters
        location: {name, latitude, longitude, countryCode?}  // same rules as a destination
        startTime: timestamp
        endTime: timestamp | null  // >= startTime
        createdByUserId: string    // auth uid of the creator
        creationTime: timestamp
        // The ID and trip come from the path. Any member can create; no updates or deletes yet.
    votes/{voteId} // later feature
        ballots/{userId}
    photos/{photoId} // later feature

// memberIds exists so "my trips" can be queried with array-contains and checked by security rules.
// The trip document and the owner member document are written in one transaction, keyed by a
// client-generated tripId so that retries are idempotent.


Realtime Database: // later feature
tripLocations/{tripId}/{userId} // later feature
presence/{userId} // later feature

Storage: // later feature
trips/{tripId}/photos/{photoId} // later feature
